function vv_sync_native_content(array $data): array {
    if(($data['format']??0)!==1||!is_array($data['items']??null)||count($data['items'])>1000||!is_string($data['source']??null))throw new RuntimeException('Invalid native snapshot.');
    $lock=fopen(__DIR__.'/vv-data/sync.lock','c+b');if(!$lock||!flock($lock,LOCK_EX))throw new RuntimeException('Sync locked.');
    try {
        global $wpdb;$seen=[];$changed=0;$created=0;
        foreach($data['items'] as $item){
            if(!is_array($item)||!preg_match('/^[a-f0-9]{24}$/D',$item['id']??'')||!in_array($item['type']??'',['post','page'],true)||!is_string($item['title']??null)||!is_string($item['body']??null)||!is_string($item['slug']??null)||!is_int($item['published']??null))throw new RuntimeException('Invalid native content item.');
            $ids=$wpdb->get_col($wpdb->prepare("SELECT p.ID FROM {$wpdb->posts} p INNER JOIN {$wpdb->postmeta} m ON p.ID=m.post_id WHERE m.meta_key='_vv_cms_id' AND m.meta_value=%s AND p.post_type IN ('post','page') ORDER BY p.ID LIMIT 2",$item['id']));
            if(count($ids)>1)throw new RuntimeException('Duplicate mirror identifier; operator repair required.');
            $id=(int)($ids[0]??0);$signature=hash('sha256',wp_json_encode($item));$seen[$item['id']]=true;
            if($id&&get_post_meta($id,'_vv_cms_signature',true)===$signature&&get_post_status($id)==='publish')continue;
            $post=['post_type'=>$item['type'],'post_status'=>'publish','post_title'=>$item['title'],'post_content'=>$item['body'],'post_excerpt'=>$item['excerpt']??'','post_name'=>$item['slug'],'post_date_gmt'=>gmdate('Y-m-d H:i:s',$item['published']),'post_date'=>get_date_from_gmt(gmdate('Y-m-d H:i:s',$item['published'])),'comment_status'=>'closed','ping_status'=>'closed'];
            if($id)$post['ID']=$id;$saved=wp_insert_post(wp_slash($post),true);if(is_wp_error($saved))throw new RuntimeException('Native mirror write failed.');
            if($id)$changed++;else$created++;update_post_meta($saved,'_vv_cms_id',$item['id']);update_post_meta($saved,'_vv_cms_source',$data['source']);update_post_meta($saved,'_vv_cms_signature',$signature);
        }
        $hidden=0;$mirrors=$wpdb->get_results("SELECT p.ID,m.meta_value FROM {$wpdb->posts} p INNER JOIN {$wpdb->postmeta} m ON p.ID=m.post_id WHERE m.meta_key='_vv_cms_id' AND p.post_type IN ('post','page') AND p.post_status='publish'");
        foreach($mirrors as $post)if(!isset($seen[$post->meta_value])&&get_post_meta($post->ID,'_vv_cms_source',true)===$data['source']){wp_update_post(['ID'=>(int)$post->ID,'post_status'=>'draft']);$hidden++;}
        if(is_array($data['site']??null)){update_option('blogname',sanitize_text_field($data['site']['title']??''));update_option('blogdescription',sanitize_text_field($data['site']['tagline']??''));}
        return ['created'=>$created,'updated'=>$changed,'hidden'=>$hidden,'source'=>$data['source']];
    } finally {flock($lock,LOCK_UN);fclose($lock);}
}
