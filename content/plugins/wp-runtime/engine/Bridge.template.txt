function vv_engine_asset_url($url) {
    if(!is_string($url))return $url;$cfg=$GLOBALS['vv_engine_config'];$base=$cfg['public_url'];
    if(!str_starts_with($url,$base.'/')&&!str_starts_with($url,'/compat/'))return $url;
    $relative=str_starts_with($url,$base.'/')?substr($url,strlen($base)+1):substr($url,8);
    $query=strpos($relative,'?');if($query!==false)$relative=substr($relative,0,$query);
    if(preg_match('~^wp-content/(themes|plugins)/([^/]+)/(.+)$~',$relative,$m))return substr($base,0,-7).'/assets/'.($m[1]==='themes'?'ui':'mod').'/'.substr(hash_hmac('sha256',$m[1].'/'.$m[2],$cfg['secret']),0,20).'/'.$m[3];
    foreach(['wp-includes/'=>'core','wp-content/uploads/'=>'media','wp-admin/'=>'admin'] as $prefix=>$alias)if(str_starts_with($relative,$prefix))return substr($base,0,-7).'/assets/'.$alias.'/'.substr($relative,strlen($prefix));
    return $url;
}
